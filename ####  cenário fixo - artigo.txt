# Geração do processo de wiener teorico X(t)
# Inicializando o processo de Wiener com valor inicial zero
wiener <- numeric(length(tempos_de_medicao))

# Gerando o processo de Wiener
for (i in 2:length(tempos_de_medicao)) {
  wiener[i]  <- rnorm(1, mean = 2 * tempos_de_medicao[i], sd = sqrt(2 * tempos_de_medicao[i]))
}

X<-wiener

# Plotando o processo de Wiener teorico
plot(tempos_de_medicao, X, type = "l", main = "Processo de Wiener", xlab = "Tempo", ylab = "Valor")
grid()
